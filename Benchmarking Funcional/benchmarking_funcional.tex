\section{Fichas de benchmarking (1 por producto)}

\subsection{Producto 1: ParkMobile}
\textbf{Plataforma:} Híbrida (web + app móvil)\\
\textbf{Audiencia objetivo:} Conductores en ciudades/campus/eventos.

\textbf{Funcionalidades relevantes (8):}
\begin{enumerate}
\item Pago de estacionamiento por zona.
\item Extensión de tiempo desde app.
\item Reserva anticipada en lotes/garajes.
\item Recordatorios de sesión.
\item Gestión de vehículos guardados.
\item Recibos/historial de pagos.
\item Pago sin efectivo/tarjeta en app.
\item Búsqueda de ubicación de estacionamiento.
\end{enumerate}

\textbf{Flujos clave:}
\begin{itemize}
\item \textit{Happy path:} buscar zona $\rightarrow$ seleccionar tiempo $\rightarrow$ pagar $\rightarrow$ recibir confirmación.
\item \textit{Alternativa:} sesión por vencer $\rightarrow$ notificación $\rightarrow$ extender tiempo (si la autoridad lo permite).
\end{itemize}

\textbf{Datos que captura/muestra:} zona, placas, método de pago, tiempo, costo total, estado de sesión, historial.

\textbf{Widgets/controles por dato:} campo de zona, selector de tiempo, botón \textit{Pagar}, botón \textit{Extender}, lista de vehículos, tarjeta de recibo.

\textbf{Evidencia y fuente:}
\begin{itemize}
\item Sitio principal: \url{https://parkmobile.io/}
\item FAQ extensión: \url{https://support.parkmobile.io/hc/en-us/articles/36854869625883-Can-I-extend-a-parking-session}
\end{itemize}

\section{Top 10 funcionalidades candidatas para el sistema (con justificación)}
\begin{enumerate}
\item \textbf{Pago móvil por zona} --- reduce filas en caseta y acelera salida.
\item \textbf{Extensión de tiempo remota} --- evita sanción por vencimiento y mejora experiencia.
\item \textbf{Confirmación digital de pago} --- facilita validación con guardia.
\item \textbf{Gestión de vehículos del usuario} --- menos errores al pagar.
\item \textbf{Historial y recibos} --- trazabilidad para soporte y auditoría.
\item \textbf{Notificaciones de vencimiento} --- previene incidencias y reclamos.
\item \textbf{Búsqueda por zona/ubicación} --- acelera selección del punto correcto.
\item \textbf{Pantalla de estado de sesión activa} --- claridad para usuario y operación.
\item \textbf{Pago mixto (digital + efectivo en contingencia)} --- robustez operativa ante fallas.
\item \textbf{Flujo de validación rápida en salida} --- menos congestión vehicular.
\end{enumerate}

\section{Decisiones de diseño}
\subsection*{3 decisiones que se adoptarán}
\begin{enumerate}
\item \textbf{Adoptar pago móvil por zona + confirmación digital.}\\
Razón: minimiza tiempos de compra y errores manuales.
\item \textbf{Adoptar notificaciones y extensión de sesión.}\\
Razón: reduce eventos de vencimiento y mejora satisfacción.
\item \textbf{Adoptar historial de transacciones y vehículos guardados.}\\
Razón: soporte más rápido y mejor trazabilidad.
\end{enumerate}

\subsection*{2 decisiones que se rechazarán}
\begin{enumerate}
\item \textbf{Rechazar dependencia exclusiva de reserva anticipada.}\\
Razón: en campus puede haber uso espontáneo; la reserva no cubre todos los casos.
\item \textbf{Rechazar flujo sin alternativa offline/efectivo.}\\
Razón: ante falla de red/terminal, se necesita contingencia para no bloquear salidas.
\end{enumerate}
